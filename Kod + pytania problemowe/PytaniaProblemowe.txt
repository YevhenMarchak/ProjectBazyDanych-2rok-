1. Ile produktów mamy w każdej kategorii, kto jest ich dostawcą, i jaka jest łączna waga produktów w tej kategorii od danego dostawcy?

SELECT 
    c.name AS category,
    s.company_name AS supplier,
    COUNT(DISTINCT p.product_id) AS total_products,
    SUM(p.weight) AS total_weight
FROM products p
JOIN categories c ON p.category_id = c.category_id
JOIN suppliers s ON p.supplier_id = s.supplier_id
JOIN inventory i ON p.product_id = i.product_id
GROUP BY c.name, s.company_name
ORDER BY c.name, s.company_name;

Cel: sprawdzenie poprawności powiązań products.category_id, products.supplier_id i inventory.product_id, użycie agregacji wielopoziomowej (COUNT DISTINCT, SUM), analiza kategorii z perspektywy dostawców i parametrów produktów.

2. Ile produktów jest dostępnych w każdym magazynie, oraz jaki procent jego maksymalnej pojemności jest zajęty?

SELECT 
    l.location_code,
    l.location_type,
    SUM(i.quantity) AS total_quantity,
    ROUND( (SUM(i.quantity) / NULLIF(l.max_capacity, 0)) * 100 , 2 ) AS capacity_usage_percent
FROM inventory i
JOIN locations l ON i.location_id = l.location_id
WHERE i.quantity > 0
GROUP BY l.location_id
ORDER BY capacity_usage_percent DESC;

Cel: sprawdzenie spójności inventory.location_id, obliczenie wykorzystania pojemności magazynu, filtrowanie danych, agregacja SUM oraz analiza relacji między stanami magazynowymi a parametrami lokalizacji.

3. Jakie zamówienia klientów są w trakcie realizacji, ile produktów mają do wysyłki oraz kiedy ostatnio zmieniono ich status (max shipment_date)?

SELECT 
    sh.shipment_id,
    c.company_name AS client,
    sh.status,
    COUNT(DISTINCT sd.product_id) AS product_count,
    SUM(sd.quantity_to_ship) AS total_quantity,
    MAX(sh.shipment_date) AS last_update
FROM shipments sh
JOIN clients c ON sh.client_id = c.client_id
JOIN shipment_details sd ON sh.shipment_id = sd.shipment_id
WHERE sh.status = 'W trakcie'
GROUP BY sh.shipment_id, c.company_name, sh.status
ORDER BY last_update DESC;

Cel: sprawdzenie relacji shipments.client_id oraz shipment_details.shipment_id, analiza zamówień według statusów, agregacja danych o liczbie produktów i ilościach do wysyłki, kontrola aktualności danych poprzez MAX(shipment_date).

4. Które dokumenty przyjęcia mają różnice między ilością oczekiwaną a faktycznie odebraną, ile produktów obejmuje ta różnica oraz jaka jest łączna suma rozbieżności?

SELECT 
    r.receipt_id,
    s.company_name AS supplier,
    COUNT(rd.product_id) AS products_with_difference,
    SUM(rd.expected_quantity - rd.received_quantity) AS difference,
    SUM(ABS(rd.expected_quantity - rd.received_quantity)) AS absolute_difference
FROM receipts r
JOIN suppliers s ON r.supplier_id = s.supplier_id
JOIN receipt_details rd ON r.receipt_id = rd.receipt_id
GROUP BY r.receipt_id, s.company_name
HAVING SUM(rd.expected_quantity - rd.received_quantity) != 0
ORDER BY absolute_difference DESC;


Cel: test integralności danych w receipt_details, rozszerzone sprawdzenie zgodności ilości oczekiwanej i odebranej, dodanie informacji o liczbie pozycji z różnicami oraz obliczenie rzeczywistej sumy rozbieżności, a także identyfikacja dokumentów z największymi odchyleniami.

5. Jakie produkty są oferowane przez ilu dostawców, w ilu magazynach występują i jaki jest całkowity zapas w firmie?

SELECT 
    p.name AS product,
    COUNT(DISTINCT p.supplier_id) AS supplier_count,
    COUNT(DISTINCT i.location_id) AS location_count,
    COALESCE(SUM(i.quantity), 0) AS total_stock
FROM products p
LEFT JOIN inventory i ON p.product_id = i.product_id
GROUP BY p.name
ORDER BY total_stock DESC;

Cel: sprawdzenie relacji między produktami, dostawcami i magazynami, wykorzystanie LEFT JOIN do uwzględnienia produktów bez stanów, agregacja liczby dostawców, liczby lokalizacji i ilości zapasów, analiza dostępności produktów.



1. Procedura
CREATE OR REPLACE PROCEDURE raport_produkty_kategorie()
LANGUAGE plpgsql
AS $$
BEGIN
    DROP TABLE IF EXISTS temp_kategorie_raport;

    CREATE TEMP TABLE temp_kategorie_raport (
        category TEXT,
        supplier TEXT,
        total_products INT,
        total_weight NUMERIC
    );

    INSERT INTO temp_kategorie_raport
    SELECT 
        c.name AS category,
        s.company_name AS supplier,
        COUNT(DISTINCT p.product_id) AS total_products,
        SUM(p.weight) AS total_weight
    FROM products p
    JOIN categories c ON p.category_id = c.category_id
    JOIN suppliers s ON p.supplier_id = s.supplier_id
    JOIN inventory i ON p.product_id = i.product_id
    GROUP BY c.name, s.company_name
    ORDER BY c.name, s.company_name;

    RAISE NOTICE 'Raport został wygenerowany %', NOW();
END;
$$;

Uruchomienie procedury:
CALL raport_produkty_kategorie();

Dane tutaj:
SELECT * FROM temp_kategorie_raport;

CREATE OR REPLACE PROCEDURE raport_zamowienia_w_trakcie()
LANGUAGE plpgsql
AS $$
DECLARE
    liczba INT;
BEGIN
    DROP TABLE IF EXISTS temp_zamowienia_trwajace;

    CREATE TEMP TABLE temp_zamowienia_trwajace (
        shipment_id INT,
        client TEXT,
        status TEXT,
        product_count INT,
        total_quantity INT,
        last_update TIMESTAMP
    );

    SELECT COUNT(*) INTO liczba FROM shipments WHERE status = 'W trakcie';
    IF liczba = 0 THEN
        RAISE NOTICE 'Brak zamówień o statusie W trakcie.';
        RETURN;
    END IF;

    INSERT INTO temp_zamowienia_trwajace
    SELECT 
        sh.shipment_id,
        c.company_name AS client,
        sh.status,
        COUNT(DISTINCT sd.product_id) AS product_count,
        SUM(sd.quantity_to_ship) AS total_quantity,
        MAX(sh.shipment_date) AS last_update
    FROM shipments sh
    JOIN clients c ON sh.client_id = c.client_id
    JOIN shipment_details sd ON sh.shipment_id = sd.shipment_id
    WHERE sh.status = 'W trakcie'
    GROUP BY sh.shipment_id, c.company_name, sh.status
    ORDER BY last_update DESC;

    RAISE NOTICE 'Raport zamówień wygenerowany: %', NOW();
END;
$$;

Uruchomienie:
CALL raport_zamowienia_w_trakcie();
SELECT * FROM temp_zamowienia_trwajace;

CREATE OR REPLACE PROCEDURE raport_rozbieznosci_przyjec()
LANGUAGE plpgsql
AS $$
BEGIN
    DROP TABLE IF EXISTS temp_rozbieznosci;

    CREATE TEMP TABLE temp_rozbieznosci (
        receipt_id INT,
        supplier TEXT,
        products_with_difference INT,
        difference NUMERIC,
        absolute_difference NUMERIC,
        severity_level TEXT
    );

    INSERT INTO temp_rozbieznosci
    SELECT 
        r.receipt_id,
        s.company_name AS supplier,
        COUNT(rd.product_id) AS products_with_difference,
        SUM(rd.expected_quantity - rd.received_quantity) AS difference,
        SUM(ABS(rd.expected_quantity - rd.received_quantity)) AS absolute_difference,
        CASE 
            WHEN SUM(ABS(rd.expected_quantity - rd.received_quantity)) > 100 THEN 'HIGH'
            WHEN SUM(ABS(rd.expected_quantity - rd.received_quantity)) > 20 THEN 'MEDIUM'
            ELSE 'LOW'
        END AS severity_level
    FROM receipts r
    JOIN suppliers s ON r.supplier_id = s.supplier_id
    JOIN receipt_details rd ON r.receipt_id = rd.receipt_id
    GROUP BY r.receipt_id, s.company_name
    HAVING SUM(rd.expected_quantity - rd.received_quantity) != 0
    ORDER BY absolute_difference DESC;

    RAISE NOTICE 'Raport rozbieżności wygenerowany: %', NOW();
END;
$$;

Uruchomienie:
CALL raport_rozbieznosci_przyjec();
SELECT * FROM temp_rozbieznosci;

CREATE OR REPLACE PROCEDURE raport_produkty_dostawy_magazyny()
LANGUAGE plpgsql
AS $$
BEGIN
    DROP TABLE IF EXISTS temp_produkty_raport;

    CREATE TEMP TABLE temp_produkty_raport (
        product TEXT,
        supplier_count INT,
        location_count INT,
        total_stock INT
    );

    INSERT INTO temp_produkty_raport
    SELECT 
        p.name AS product,
        COUNT(DISTINCT p.supplier_id) AS supplier_count,
        COUNT(DISTINCT i.location_id) AS location_count,
        COALESCE(SUM(i.quantity), 0) AS total_stock
    FROM products p
    LEFT JOIN inventory i ON p.product_id = i.product_id
    GROUP BY p.name
    ORDER BY total_stock DESC;

    RAISE NOTICE 'Raport produktów wygenerowany: %', NOW();
END;
$$;
